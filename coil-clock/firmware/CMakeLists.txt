cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(coil_clock C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()

target_link_libraries(coil_clock pico_stdlib hardware_adc)
pico_enable_stdio_usb(coil_clock 1)
pico_enable_stdio_uart(coil_clock 0)
pico_add_extra_outputs(coil_clock)
