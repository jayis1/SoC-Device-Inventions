cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(dial_reader C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(dial_reader src/main.c)
target_include_directories(dial_reader PRIVATE include)
target_link_libraries(dial_reader pico_stdlib hardware_adc m)
pico_enable_stdio_usb(dial_reader 1)
pico_enable_stdio_uart(dial_reader 0)
pico_add_extra_outputs(dial_reader)
