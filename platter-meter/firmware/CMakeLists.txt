cmake_minimum_required(VERSION 3.20)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(platter_meter C CXX ASM)
pico_sdk_init()
add_executable(platter_meter src/main.c src/platter_meter.c)
target_link_libraries(platter_meter pico_stdlib hardware_gpio)
pico_enable_stdio_usb(platter_meter 1)
pico_enable_stdio_uart(platter_meter 0)
pico_add_extra_outputs(platter_meter)
