cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(flex_cycle C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(flex_cycle src/main.c)
target_link_libraries(flex_cycle pico_stdlib hardware_gpio)
pico_enable_stdio_usb(flex_cycle 1)
pico_enable_stdio_uart(flex_cycle 0)
pico_add_extra_outputs(flex_cycle)
