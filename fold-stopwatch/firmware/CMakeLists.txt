cmake_minimum_required(VERSION 3.16)
project(fold_stopwatch_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(fold_stopwatch src/fold_stopwatch.c)
target_include_directories(fold_stopwatch PUBLIC include)
add_executable(test_fold_stopwatch tests/test_fold_stopwatch.c)
target_link_libraries(test_fold_stopwatch PRIVATE fold_stopwatch)
enable_testing()
add_test(NAME fold_stopwatch_core COMMAND test_fold_stopwatch)
