set(PICO_BOARD pico)
set(PICO_PLATFORM rp2040)
cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(morse_mate C CXX ASM)
pico_sdk_init()
add_executable(morse_mate main.c)
target_link_libraries(morse_mate pico_stdlib hardware_gpio hardware_pwm)
pico_enable_stdio_usb(morse_mate 1)
pico_enable_stdio_uart(morse_mate 0)
pico_add_extra_outputs(morse_mate)
