cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(film_gate C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(film_gate src/main.c)
target_link_libraries(film_gate pico_stdlib)
pico_enable_stdio_usb(film_gate 1)
pico_enable_stdio_uart(film_gate 0)
pico_add_extra_outputs(film_gate)
