cmake_minimum_required(VERSION 3.20)
project(curb_cue_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(curb_cue_core src/curb_cue.c)
target_include_directories(curb_cue_core PUBLIC include)
target_compile_options(curb_cue_core PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(curb_cue_demo src/main.c)
target_link_libraries(curb_cue_demo PRIVATE curb_cue_core)
add_executable(test_curb_cue tests/test_curb_cue.c)
target_link_libraries(test_curb_cue PRIVATE curb_cue_core)
enable_testing()
add_test(NAME curb_cue_policy COMMAND test_curb_cue)
