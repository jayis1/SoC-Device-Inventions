cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(resistor_lens C CXX ASM)
pico_sdk_init()
add_executable(resistor_lens main.c)
target_link_libraries(resistor_lens pico_stdlib hardware_i2c)
pico_enable_stdio_usb(resistor_lens 1)
pico_enable_stdio_uart(resistor_lens 0)
pico_add_extra_outputs(resistor_lens)
