cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(knot_witness C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(knot_witness main.c core.c)
target_include_directories(knot_witness PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(knot_witness pico_stdlib)
pico_enable_stdio_usb(knot_witness 1)
pico_enable_stdio_uart(knot_witness 0)
pico_add_extra_outputs(knot_witness)
