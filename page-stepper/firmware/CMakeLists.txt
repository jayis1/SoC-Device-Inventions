cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(page_stepper C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(page_stepper main.c usb_descriptors.c)
target_include_directories(page_stepper PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(page_stepper pico_stdlib tinyusb_device tinyusb_board)
pico_enable_stdio_usb(page_stepper 0)
pico_enable_stdio_uart(page_stepper 0)
pico_add_extra_outputs(page_stepper)
