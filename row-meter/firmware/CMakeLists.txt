cmake_minimum_required(VERSION 3.16)
project(row_meter_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(row_meter_core src/row_meter.c)
target_include_directories(row_meter_core PUBLIC include)
add_executable(test_row_meter tests/test_row_meter.c)
target_link_libraries(test_row_meter PRIVATE row_meter_core)
enable_testing()
add_test(NAME row_meter_core COMMAND test_row_meter)
