cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(tide_tab C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(tide_tab src/main.c src/tide_tab.c)
target_link_libraries(tide_tab pico_stdlib)
pico_enable_stdio_usb(tide_tab 1)
pico_enable_stdio_uart(tide_tab 0)
pico_add_extra_outputs(tide_tab)
