cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(reed_ranger C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(reed_ranger src/main.c)
target_link_libraries(reed_ranger pico_stdlib)
pico_enable_stdio_usb(reed_ranger 1)
pico_enable_stdio_uart(reed_ranger 0)
pico_add_extra_outputs(reed_ranger)
