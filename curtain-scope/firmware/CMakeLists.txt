cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(curtain_scope C CXX ASM)
pico_sdk_init()
add_executable(curtain_scope src/main.c)
target_include_directories(curtain_scope PRIVATE include)
target_link_libraries(curtain_scope pico_stdlib hardware_gpio hardware_timer)
pico_enable_stdio_usb(curtain_scope 1)
pico_enable_stdio_uart(curtain_scope 0)
pico_add_extra_outputs(curtain_scope)
