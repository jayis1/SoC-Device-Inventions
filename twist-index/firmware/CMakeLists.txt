cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(twist_index C CXX ASM)
pico_sdk_init()
add_executable(twist_index src/main.c src/core.c)
target_include_directories(twist_index PRIVATE include)
target_link_libraries(twist_index pico_stdlib)
pico_enable_stdio_usb(twist_index 1)
pico_enable_stdio_uart(twist_index 0)
pico_add_extra_outputs(twist_index)
