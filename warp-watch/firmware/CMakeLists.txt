cmake_minimum_required(VERSION 3.20)
project(warp_watch_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(warp_watch_core src/warp_watch.c)
target_include_directories(warp_watch_core PUBLIC include)
target_compile_options(warp_watch_core PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(warp_watch_demo src/main.c)
target_link_libraries(warp_watch_demo PRIVATE warp_watch_core)
add_executable(test_warp_watch tests/test_warp_watch.c)
target_link_libraries(test_warp_watch PRIVATE warp_watch_core)
enable_testing()
add_test(NAME warp_watch_policy COMMAND test_warp_watch)
