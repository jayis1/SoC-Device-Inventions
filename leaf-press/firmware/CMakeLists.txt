cmake_minimum_required(VERSION 3.16)
project(leaf_press_core C)
set(CMAKE_C_STANDARD 11)
add_library(leaf_press leaf_press.c)
target_compile_options(leaf_press PRIVATE -Wall -Wextra -Werror)
add_executable(test_leaf_press test_leaf_press.c)
target_link_libraries(test_leaf_press PRIVATE leaf_press)
enable_testing()
add_test(NAME core COMMAND test_leaf_press)
