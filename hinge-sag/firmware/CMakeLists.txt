cmake_minimum_required(VERSION 3.16)
project(hinge_sag_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(hinge_sag_core src/hinge_sag.c)
target_include_directories(hinge_sag_core PUBLIC include)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_hinge_sag tests/test_hinge_sag.c)
  target_link_libraries(test_hinge_sag PRIVATE hinge_sag_core m)
  add_test(NAME hinge_sag_core COMMAND test_hinge_sag)
endif()
