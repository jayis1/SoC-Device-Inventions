cmake_minimum_required(VERSION 3.16)
project(cargo_canary C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_library(cargo_canary cargo_canary.c)
target_include_directories(cargo_canary PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(cargo_canary_demo main.c)
target_link_libraries(cargo_canary_demo PRIVATE cargo_canary)
add_executable(test_cargo_canary test_cargo_canary.c)
target_link_libraries(test_cargo_canary PRIVATE cargo_canary)
add_test(NAME cargo_canary_core COMMAND test_cargo_canary)
