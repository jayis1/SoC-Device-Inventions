cmake_minimum_required(VERSION 3.16)
project(selvage_scout_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(scout_core core.c)
target_include_directories(scout_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(test_core test_core.c)
target_link_libraries(test_core PRIVATE scout_core)
enable_testing()
add_test(NAME core COMMAND test_core)
