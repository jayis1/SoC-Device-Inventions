cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(clay_ledger C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(clay_ledger src/main.c)
target_link_libraries(clay_ledger pico_stdlib hardware_i2c)
pico_enable_stdio_usb(clay_ledger 1)
pico_enable_stdio_uart(clay_ledger 0)
pico_add_extra_outputs(clay_ledger)
