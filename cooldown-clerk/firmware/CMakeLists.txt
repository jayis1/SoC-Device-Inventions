cmake_minimum_required(VERSION 3.16)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(cooldown_clerk C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(cooldown_clerk src/main.c)
target_link_libraries(cooldown_clerk pico_stdlib hardware_i2c)
pico_enable_stdio_usb(cooldown_clerk 1)
pico_enable_stdio_uart(cooldown_clerk 0)
pico_add_extra_outputs(cooldown_clerk)
