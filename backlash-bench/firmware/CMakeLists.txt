cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(backlash_bench C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(backlash_bench main.c)
target_link_libraries(backlash_bench pico_stdlib hardware_i2c)
pico_enable_stdio_usb(backlash_bench 1)
pico_enable_stdio_uart(backlash_bench 0)
pico_add_extra_outputs(backlash_bench)
