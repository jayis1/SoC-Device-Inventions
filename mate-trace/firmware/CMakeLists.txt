cmake_minimum_required(VERSION 3.16)
project(mate_trace_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(mate_trace STATIC src/mate_trace.c)
target_include_directories(mate_trace PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_options(mate_trace PRIVATE -Wall -Wextra -Werror -pedantic)
enable_testing()
add_executable(test_mate_trace test/test_mate_trace.c)
target_compile_options(test_mate_trace PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(test_mate_trace PRIVATE mate_trace)
add_test(NAME mate_trace_core COMMAND test_mate_trace)
