cmake_minimum_required(VERSION 3.16)
project(patch_card_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(measurement_core src/measurement.c)
target_include_directories(measurement_core PUBLIC include)
target_compile_options(measurement_core PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(test_measurement tests/test_measurement.c)
target_link_libraries(test_measurement PRIVATE measurement_core)
enable_testing()
add_test(NAME measurement COMMAND test_measurement)
