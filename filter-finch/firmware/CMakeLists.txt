cmake_minimum_required(VERSION 3.16)
project(filter_finch C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Werror -pedantic)
add_library(filter_core filter_core.c)
target_include_directories(filter_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(filter_core PUBLIC m)
add_executable(filter_finch_demo main.c)
target_link_libraries(filter_finch_demo PRIVATE filter_core)
add_executable(test_filter_core test_filter_core.c)
target_link_libraries(test_filter_core PRIVATE filter_core)
enable_testing()
add_test(NAME filter_core COMMAND test_filter_core)
