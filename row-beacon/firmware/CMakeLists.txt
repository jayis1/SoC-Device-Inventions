cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(row_beacon C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(row_beacon main.c row_core.c)
target_include_directories(row_beacon PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(row_beacon pico_stdlib)
pico_add_extra_outputs(row_beacon)
