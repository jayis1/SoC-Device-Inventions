cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(darkroom_dose C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(darkroom_dose src/main.c)
target_link_libraries(darkroom_dose pico_stdlib hardware_adc)
pico_enable_stdio_usb(darkroom_dose 1)
pico_enable_stdio_uart(darkroom_dose 0)
pico_add_extra_outputs(darkroom_dose)
