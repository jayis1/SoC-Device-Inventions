cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(paper_dryness C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(paper_dryness main.c)
target_link_libraries(paper_dryness pico_stdlib hardware_spi)
pico_enable_stdio_usb(paper_dryness 1)
pico_enable_stdio_uart(paper_dryness 0)
pico_add_extra_outputs(paper_dryness)
