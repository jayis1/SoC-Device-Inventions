cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(loom_latch C CXX ASM)
pico_sdk_init()
add_executable(loom_latch main.c classifier.c)
target_include_directories(loom_latch PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(loom_latch pico_stdlib)
pico_enable_stdio_usb(loom_latch 1)
pico_enable_stdio_uart(loom_latch 0)
pico_add_extra_outputs(loom_latch)
