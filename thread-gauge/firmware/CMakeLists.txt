cmake_minimum_required(VERSION 3.16)
project(thread_gauge_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(thread_gauge_core STATIC main.c)
add_executable(thread_gauge_test test.c)

enable_testing()
add_test(NAME conversion COMMAND thread_gauge_test)
