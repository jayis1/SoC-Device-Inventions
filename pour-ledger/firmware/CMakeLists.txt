cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(pour_ledger C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(pour_ledger src/main.c)
target_link_libraries(pour_ledger pico_stdlib hardware_pio hardware_i2c)
pico_enable_stdio_usb(pour_ledger 1)
pico_enable_stdio_uart(pour_ledger 0)
pico_add_extra_outputs(pour_ledger)
