cmake_minimum_required(VERSION 3.16)
project(wheel_whisper C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(CTest)
add_library(wheel_whisper_core wheel_whisper.c)
target_include_directories(wheel_whisper_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(wheel_whisper_core PUBLIC m)
add_executable(wheel_whisper_demo main.c)
target_link_libraries(wheel_whisper_demo PRIVATE wheel_whisper_core)
add_executable(test_wheel_whisper test_wheel_whisper.c)
target_link_libraries(test_wheel_whisper PRIVATE wheel_whisper_core)
add_test(NAME wheel_whisper_core COMMAND test_wheel_whisper)
