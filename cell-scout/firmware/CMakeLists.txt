cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(cell_scout C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(cell_scout main.c)
target_link_libraries(cell_scout pico_stdlib hardware_adc)
pico_enable_stdio_usb(cell_scout 1)
pico_enable_stdio_uart(cell_scout 0)
pico_add_extra_outputs(cell_scout)
