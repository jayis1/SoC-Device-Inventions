cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(tea_steep C CXX ASM)
pico_sdk_init()
add_executable(tea_steep src/main.c src/telemetry.c)
target_include_directories(tea_steep PRIVATE include)
target_link_libraries(tea_steep pico_stdlib hardware_i2c)
pico_enable_stdio_usb(tea_steep 1)
pico_enable_stdio_uart(tea_steep 0)
pico_add_extra_outputs(tea_steep)
