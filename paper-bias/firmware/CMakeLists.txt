cmake_minimum_required(VERSION 3.16)
project(paper_bias_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(paper_bias_core src/paper_bias.c)
target_include_directories(paper_bias_core PUBLIC include)
target_compile_options(paper_bias_core PRIVATE -Wall -Wextra -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(paper_bias_test test_core.c)
  target_link_libraries(paper_bias_test PRIVATE paper_bias_core)
  add_test(NAME paper_bias_core_test COMMAND paper_bias_test)
endif()
