cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(crease_checker C CXX ASM)
pico_sdk_init()
add_executable(crease-checker main.c core.c)
target_include_directories(crease-checker PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(crease-checker pico_stdlib hardware_i2c)
pico_enable_stdio_usb(crease-checker 1)
pico_enable_stdio_uart(crease-checker 0)
pico_add_extra_outputs(crease-checker)
