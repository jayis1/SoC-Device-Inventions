cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(warp_meter C CXX ASM)
pico_sdk_init()
add_executable(warp_meter main.c)
target_link_libraries(warp_meter pico_stdlib hardware_gpio)
pico_enable_stdio_usb(warp_meter 1)
pico_enable_stdio_uart(warp_meter 0)
pico_add_extra_outputs(warp_meter)
