cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(burette_witness C CXX ASM)
pico_sdk_init()
add_executable(burette_witness src/main.c src/logger.c)
target_include_directories(burette_witness PRIVATE src)
target_link_libraries(burette_witness pico_stdlib hardware_i2c)
pico_enable_stdio_usb(burette_witness 1)
pico_enable_stdio_uart(burette_witness 0)
pico_add_extra_outputs(burette_witness)
