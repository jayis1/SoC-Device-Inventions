cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(fret_meter C CXX ASM)
pico_sdk_init()
add_executable(fret_meter src/main.c src/fret_meter.c)
target_include_directories(fret_meter PRIVATE include)
target_link_libraries(fret_meter pico_stdlib hardware_adc)
pico_enable_stdio_usb(fret_meter 1)
pico_enable_stdio_uart(fret_meter 0)
pico_add_extra_outputs(fret_meter)
