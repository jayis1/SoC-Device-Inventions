cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(shutter_tick C CXX ASM)
pico_sdk_init()
add_executable(shutter_tick main.c timing.c)
target_link_libraries(shutter_tick pico_stdlib)
pico_enable_stdio_usb(shutter_tick 1)
pico_enable_stdio_uart(shutter_tick 0)
target_include_directories(shutter_tick PRIVATE ${CMAKE_CURRENT_LIST_DIR})
pico_add_extra_outputs(shutter_tick)
