cmake_minimum_required(VERSION 3.16)
project(shade_scout C)
set(CMAKE_C_STANDARD 11)
option(SHADE_SCOUT_HOST_TEST "Build portable formatter test" OFF)
if(SHADE_SCOUT_HOST_TEST)
  enable_testing()
  add_executable(shade_scout_test src/main.c)
  target_compile_definitions(shade_scout_test PRIVATE SHADE_SCOUT_TEST)
  add_test(NAME formatter COMMAND shade_scout_test)
else()
  if(NOT DEFINED PICO_SDK_PATH)
    message(FATAL_ERROR "Set PICO_SDK_PATH to Raspberry Pi Pico SDK 2.1.1")
  endif()
  include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
  pico_sdk_init()
  add_executable(shade_scout src/main.c src/pico_main.c)
  target_link_libraries(shade_scout pico_stdlib hardware_i2c)
  pico_enable_stdio_usb(shade_scout 1)
  pico_enable_stdio_uart(shade_scout 0)
  pico_add_extra_outputs(shade_scout)
endif()
