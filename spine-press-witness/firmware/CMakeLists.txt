cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(spine_press_witness C CXX ASM)
pico_sdk_init()
add_executable(spine_press_witness main.c)
target_link_libraries(spine_press_witness pico_stdlib)
pico_enable_stdio_usb(spine_press_witness 1)
pico_enable_stdio_uart(spine_press_witness 0)
pico_add_extra_outputs(spine_press_witness)
target_compile_options(spine_press_witness PRIVATE -Wall -Wextra -Werror)
