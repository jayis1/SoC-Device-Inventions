cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(tank_tally C CXX ASM)
pico_sdk_init()
add_executable(tank_tally main.c)
target_include_directories(tank_tally PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(tank_tally pico_stdlib hardware_i2c)
pico_enable_stdio_usb(tank_tally 1)
pico_enable_stdio_uart(tank_tally 0)
pico_add_extra_outputs(tank_tally)
