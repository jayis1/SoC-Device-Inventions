cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(key_travel C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(key_travel main.c)
target_link_libraries(key_travel pico_stdlib hardware_i2c)
pico_enable_stdio_usb(key_travel 1)
pico_enable_stdio_uart(key_travel 0)
pico_add_extra_outputs(key_travel)
