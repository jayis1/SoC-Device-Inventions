cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(latch_ledger C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(latch_ledger main.c)
target_link_libraries(latch_ledger pico_stdlib)
pico_enable_stdio_usb(latch_ledger 1)
pico_enable_stdio_uart(latch_ledger 0)
pico_add_extra_outputs(latch_ledger)
