cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(wick_watch C CXX ASM)
pico_sdk_init()
add_executable(wick_watch src/main.c)
target_link_libraries(wick_watch pico_stdlib)
pico_enable_stdio_usb(wick_watch 1)
pico_enable_stdio_uart(wick_watch 0)
pico_add_extra_outputs(wick_watch)
target_compile_options(wick_watch PRIVATE -Wall -Wextra -Werror)
