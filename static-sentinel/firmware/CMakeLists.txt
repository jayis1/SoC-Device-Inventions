cmake_minimum_required(VERSION 3.16)
project(static_sentinel C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra -Werror -pedantic")
add_library(esd_core src/esd_core.c)
target_include_directories(esd_core PUBLIC include)
target_link_libraries(esd_core m)
add_executable(static_sentinel_demo src/main.c)
target_link_libraries(static_sentinel_demo esd_core)
add_executable(test_esd_core tests/test_esd_core.c)
target_link_libraries(test_esd_core esd_core)
enable_testing()
add_test(NAME esd_core COMMAND test_esd_core)
