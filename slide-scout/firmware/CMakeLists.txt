cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(slide_scout C CXX ASM)
pico_sdk_init()
add_executable(slide_scout main.c density.c)
target_link_libraries(slide_scout pico_stdlib hardware_i2c)
pico_enable_stdio_usb(slide_scout 1)
pico_enable_stdio_uart(slide_scout 0)
pico_add_extra_outputs(slide_scout)
target_include_directories(slide_scout PRIVATE ${CMAKE_CURRENT_LIST_DIR})
