cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(flow_ledger C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()

target_link_libraries(flow_ledger pico_stdlib)
pico_enable_stdio_usb(flow_ledger 1)
pico_enable_stdio_uart(flow_ledger 0)
pico_add_extra_outputs(flow_ledger)
