cmake_minimum_required(VERSION 3.16)
project(braille_check_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(braille_core src/measurement.c)
target_include_directories(braille_core PUBLIC include)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_measurement tests/test_measurement.c)
  target_link_libraries(test_measurement PRIVATE braille_core)
  add_test(NAME measurement COMMAND test_measurement)
endif()
