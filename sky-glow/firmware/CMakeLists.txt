cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(sky_glow C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(sky_glow src/main.c src/sky_math.c)
target_include_directories(sky_glow PRIVATE include)
target_link_libraries(sky_glow pico_stdlib hardware_i2c)
pico_enable_stdio_usb(sky_glow 1)
pico_enable_stdio_uart(sky_glow 0)
pico_add_extra_outputs(sky_glow)
